+"""SQLite storage for TeamBoard tasks."""
+import sqlite3
+
+
+def connect(path=":memory:"):
+    """Open a database and make sure the tasks table exists."""
+    conn = sqlite3.connect(path)
+    conn.row_factory = sqlite3.Row
+    conn.execute(
+        """CREATE TABLE IF NOT EXISTS tasks (
+               id       INTEGER PRIMARY KEY AUTOINCREMENT,
+               title    TEXT NOT NULL,
+               assignee TEXT,
+               due      TEXT NOT NULL,
+               done     INTEGER NOT NULL DEFAULT 0
+           )"""
+    )
+    return conn
+
+
+def add_task(conn, title, assignee, due):
+    """Insert a task. `due` is an ISO date string such as '2026-11-23'. Returns the new id."""
+    cur = conn.execute(
+        "INSERT INTO tasks (title, assignee, due) VALUES (?, ?, ?)",
+        (title, assignee, due),
+    )
+    conn.commit()
+    return cur.lastrowid
+
+
+def mark_done(conn, task_id):
+    """Mark one task as done."""
+    conn.execute("UPDATE tasks SET done = 1 WHERE id = ?", (task_id,))
+    conn.commit()
+
+
+def all_tasks(conn):
+    """Return every task as a dict, soonest due date first."""
+    rows = conn.execute("SELECT * FROM tasks ORDER BY due, id")
+    return [dict(r) for r in rows]
+
+
+def find_by_assignee(conn, name):
+    """Return all tasks assigned to `name`, soonest due date first."""
+    query = f"SELECT * FROM tasks WHERE assignee = '{name}' ORDER BY due, id"
+    return [dict(r) for r in conn.execute(query)]

+"""Plain-text reports for the team."""
+import time
+
+from teamboard.tasks import due_within
+
+
+def workload_summary(tasks):
+    """Count open tasks per person. Tasks with no assignee are counted under 'Unassigned'."""
+    counts = {}
+    for task in tasks:
+        if task["done"]:
+            continue
+        name = task["assignee"].strip().title()
+        counts[name] = counts.get(name, 0) + 1
+    return counts
+
+
+def timed_report(tasks):
+    """Return the workload report followed by a line saying how long it took to build."""
+    start = time.clock()
+    summary = workload_summary(tasks)
+    elapsed = time.clock() - start
+    lines = [f"{name}: {count} open" for name, count in sorted(summary.items())]
+    lines.append(f"(report built in {elapsed * 1000:.2f} ms)")
+    return "\n".join(lines)
+
+
+def due_soon_report(tasks, today, days=3):
+    """List the open tasks due within `days` days."""
+    soon = due_within(tasks, today, days)
+    if not soon:
+        return "Nothing due soon."
+    return "\n".join(f"{t['due']}  {t['title']} ({t['assignee'] or 'unassigned'})" for t in soon)

+"""Pure functions that work on lists of task dicts (as returned by db.all_tasks)."""
+from datetime import date
+
+
+def parse_due(text):
+    """Turn '2026-11-23' into a date."""
+    return date.fromisoformat(text)
+
+
+def due_within(tasks, today, days):
+    """Return the open tasks due from `today` up to and including `days` days from today.
+
+    Example: with days=3, a task due exactly 3 days from today IS included.
+    Finished tasks and tasks already overdue are not included.
+    """
+    result = []
+    for task in tasks:
+        if task["done"]:
+            continue
+        delta = (parse_due(task["due"]) - today).days
+        if 0 <= delta < days:
+            result.append(task)
+    return result
+
+
+def find_duplicate_titles(tasks):
+    """Return the titles that appear more than once, ignoring case and spaces.
+
+    Each duplicated title is returned once, in lower case, in the order it first repeats.
+    """
+    seen = []
+    duplicates = []
+    for task in tasks:
+        title = task["title"].strip().lower()
+        if title in seen:
+            if title not in duplicates:
+                duplicates.append(title)
+        else:
+            seen.append(title)
+    return duplicates

+    return conn

+class StorageTests(unittest.TestCase):
+    def test_add_and_list_tasks(self):
+        conn = make_db()
+        titles = [t["title"] for t in db.all_tasks(conn)]
+        self.assertEqual(titles, ["Fix login bug", "Write report", "Prepare demo"])
+
+    def test_mark_done(self):
+        conn = make_db()
+        first = db.all_tasks(conn)[0]

+    def test_find_by_assignee_returns_that_persons_tasks(self):

+class DueDateTests(unittest.TestCase):
+    TODAY = date(2026, 11, 18)
+
+    def test_due_within_includes_the_last_day(self):
+        conn = make_db()
+        db.add_task(conn, "Boundary task", "cara wu", "2026-11-21")  # exactly 3 days away
+        soon = tasks.due_within(db.all_tasks(conn), self.TODAY, 3)
+        self.assertIn("Boundary task", [t["title"] for t in soon])
+
+    def test_due_within_skips_done_and_past(self):
+        conn = make_db()

+        db.mark_done(conn, done_id)
